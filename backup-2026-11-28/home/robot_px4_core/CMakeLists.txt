cmake_minimum_required(VERSION 3.20)
project(robot_px4_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
include(FetchContent)
FetchContent_Declare(rplidar_sdk GIT_REPOSITORY https://github.com/Slamtec/rplidar_sdk.git GIT_TAG 99478e5fb90de3b4a6db0080acacd373f8b36869)
FetchContent_MakeAvailable(rplidar_sdk)
set(s ${rplidar_sdk_SOURCE_DIR}/sdk)
file(GLOB sdk_src CONFIGURE_DEPENDS "${s}/src/*.cpp" "${s}/src/arch/linux/*.cpp" "${s}/src/hal/*.cpp")
file(GLOB_RECURSE unpack CONFIGURE_DEPENDS "${s}/src/dataunpacker/*.cpp")
add_library(slamtec STATIC ${sdk_src} ${unpack})
target_include_directories(slamtec PUBLIC ${s}/include PRIVATE ${s}/src)
target_link_libraries(slamtec PUBLIC Threads::Threads)
add_executable(diag_sensors src/app/diag_sensors.cpp src/drivers/hwt905_driver.cpp src/drivers/rplidar_driver.cpp)
target_include_directories(diag_sensors PRIVATE src)
target_link_libraries(diag_sensors PRIVATE slamtec)
add_executable(test_hwt905_parser tests/test_hwt905_parser.cpp)
target_include_directories(test_hwt905_parser PRIVATE src)
enable_testing()
add_test(NAME hwt905_parser COMMAND test_hwt905_parser)
